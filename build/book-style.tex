%% =================================================================
%%  Small Teams, Strong Systems — Print-Ready Book Design
%%  Applied via --include-in-header; zero content changes required.
%% =================================================================

%% Better typography (microtypography, ligatures, kerning)
\usepackage{microtype}

%% Widow and orphan control — no lone lines at top or bottom of pages
\widowpenalty=10000
\clubpenalty=10000
\brokenpenalty=4991

%% Heading design — title is auto-appended in format font (no [explicit] needed)
\usepackage{titlesec}

%% Part pages  (# heading → \chapter in book class)
%% Centred with generous white space — clean separator page
\titleformat{\chapter}[display]
  {\normalfont\centering\sffamily}
  {}
  {0pt}
  {\vspace{3\baselineskip}\Huge\bfseries}
  [\vspace{2\baselineskip}]

\titlespacing*{\chapter}{0pt}{0pt}{0pt}

%% Chapter openings (## heading → \section)
%% Large sans-serif, flush left — clear hierarchy
%% No {#1} here: title is auto-appended after the (empty) before-code
\titleformat{\section}
  {\normalfont\LARGE\bfseries\sffamily}
  {}
  {0pt}
  {}

\titlespacing*{\section}{0pt}{3\baselineskip}{1.5\baselineskip}

%% Sub-section headings (### → \subsection)
\titleformat{\subsection}
  {\normalfont\large\bfseries\sffamily}
  {}
  {0pt}
  {}

\titlespacing*{\subsection}{0pt}{1.2\baselineskip}{0.4\baselineskip}

%% Sub-sub-section (#### → \subsubsection)
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\sffamily}
  {}
  {0pt}
  {}

\titlespacing*{\subsubsection}{0pt}{0.8\baselineskip}{0.2\baselineskip}

%% Force every chapter opening (## = \section in book class) onto a fresh page.
\usepackage{etoolbox}
\pretocmd{\section}{\clearpage}{}{}

%% fancyhdr — load explicitly so this file is self-contained.
%% metadata.yaml also loads it; \usepackage is idempotent (second call is a no-op).
\usepackage{fancyhdr}

%% Running headers — this \AtBeginDocument runs AFTER metadata.yaml's hook,
%% so it overrides the plain centered-page-number style with a proper book layout.
\AtBeginDocument{%
  %% Suppress part-page chapter marks in the header
  \renewcommand{\chaptermark}[1]{\markboth{}{}}%
  %% Capture section (chapter) title for right-hand running header
  \renewcommand{\sectionmark}[1]{\markright{#1}}%
  %%
  \pagestyle{fancy}%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0.3pt}%
  %% Even pages (left side when open): book title
  \fancyhead[LE]{\small\itshape Small Teams, Strong Systems}%
  %% Odd pages (right side when open): current chapter title
  \fancyhead[RO]{\small\itshape\rightmark}%
  %% Page numbers in outer corners — standard print-book convention
  \fancyfoot[LE,RO]{\small\thepage}%
  \fancyfoot[CE,CO]{}%
  %% Opening pages of chapters: no header rule, page number still shows
  \fancypagestyle{plain}{%
    \fancyhf{}%
    \renewcommand{\headrulewidth}{0pt}%
    \fancyfoot[LE,RO]{\small\thepage}%
  }%
  \setlength{\headheight}{14pt}%
}
