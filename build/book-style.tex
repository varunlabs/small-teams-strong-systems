%% Better typography
\usepackage{microtype}

%% Suppress automatic part/chapter numbering. Every heading already carries its
%% own label in the text ("Part I — ...", "Chapter 1 — ..."), so LaTeX's auto
%% "Chapter 1" line above the title would be a redundant duplicate. (No effect on
%% the v2 build, whose headings are already unnumbered.)
\setcounter{secnumdepth}{-2}

%% Widow and orphan control
\widowpenalty=10000
\clubpenalty=10000

%% Sections (## headings) flow naturally within a chapter, matching the
%% published reference PDF. Chapters (# headings) still open a fresh page
%% via the book class's native \chapter, so structure is preserved without
%% forcing every sub-section onto its own page (which left large white gaps).
\raggedbottom

%% Clean, consistent page furniture — matches the reference PDF.
%% The book class's default two-sided "headings" style put a running
%% chapter/CONTENTS header at the top of every page and alternated the page
%% number between the top-left and top-right corners. Replace it with a plain
%% style: no running header, page number centered at the foot of every page.
\usepackage{fancyhdr}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
  \fancyfoot[C]{\thepage}%
}
\pagestyle{plain}
\AtBeginDocument{\pagestyle{plain}}
