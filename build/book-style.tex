%% =================================================================
%%  Small Teams, Strong Systems — Print-Ready Book Design
%%  Applied via --include-in-header; zero content changes required.
%% =================================================================

%% Better typography (microtypography, ligatures, kerning)
\usepackage{microtype}

%% Widow and orphan control — no lone lines at top or bottom of pages
\widowpenalty=10000
\clubpenalty=10000
\brokenpenalty=4991

%% Heading design
\usepackage[explicit]{titlesec}

%% Part pages  (# heading → \chapter in book class)
%% Centred vertically-ish with generous white space — clean separator page
\titleformat{\chapter}[display]
  {\normalfont\centering\sffamily}
  {}
  {0pt}
  {\vspace{3\baselineskip}\Huge\bfseries}
  [\vspace{2\baselineskip}]

\titlespacing*{\chapter}{0pt}{0pt}{0pt}

%% Chapter openings (## heading → \section)
%% Large sans-serif, flush left — clear hierarchy
\titleformat{\section}
  {\normalfont\LARGE\bfseries\sffamily}
  {}
  {0pt}
  {#1}

\titlespacing*{\section}{0pt}{0pt}{1.5\baselineskip}

%% Sub-section headings (### → \subsection)
\titleformat{\subsection}
  {\normalfont\large\bfseries\sffamily}
  {}
  {0pt}
  {#1}

\titlespacing*{\subsection}{0pt}{1.2\baselineskip}{0.4\baselineskip}

%% Sub-sub-section (#### → \subsubsection)
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\sffamily}
  {}
  {0pt}
  {#1}

\titlespacing*{\subsubsection}{0pt}{0.8\baselineskip}{0.2\baselineskip}

%% Force every chapter (## heading = \section) to start on a fresh page.
%% This must be in the preamble, not in \AtBeginDocument.
\usepackage{etoolbox}
\pretocmd{\section}{\clearpage}{}{}

%% Running headers — overrides metadata.yaml's \AtBeginDocument setup
%% by queuing a second \AtBeginDocument that runs after it.
\AtBeginDocument{%
  %% Store chapter names silently (Part pages — we don't show them in headers)
  \renewcommand{\chaptermark}[1]{\markboth{}{}}%
  %% Store section (chapter) name for right-hand running header
  \renewcommand{\sectionmark}[1]{\markright{#1}}%
  %%
  \pagestyle{fancy}%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0.3pt}%
  %% Even pages (left side when book is open): book title
  \fancyhead[LE]{\small\itshape Small Teams, Strong Systems}%
  %% Odd pages (right side when book is open): current chapter title
  \fancyhead[RO]{\small\itshape\rightmark}%
  %% Page numbers in outer corners — standard for print books
  \fancyfoot[LE,RO]{\small\thepage}%
  \fancyfoot[CE,CO]{}%
  %% Part/chapter opening pages: no header, page number still in outer corner
  \fancypagestyle{plain}{%
    \fancyhf{}%
    \renewcommand{\headrulewidth}{0pt}%
    \fancyfoot[LE,RO]{\small\thepage}%
  }%
  \setlength{\headheight}{14pt}%
}
